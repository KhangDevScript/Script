    <script>
        function activate(name, value) {
            console.log(`Activated: ${name} | Value: ${value}`);
            
            // Visual Feedback
            const btn = event.currentTarget;
            const originalText = btn.innerHTML;
            
            btn.innerHTML = `<span>Active</span><span class="badge">✔</span>`;
            btn.style.backgroundColor = "#005a9e";
            
            setTimeout(() => {
                btn.innerHTML = originalText;
                btn.style.backgroundColor = "#333";
            }, 500);
        }
