local DOUBLE_MULTIPLIER = 2

ButtonFrame.Position = UDim2.new(0, 10, 0, 30)
ButtonFrame.BackgroundTransparency = 1
ButtonFrame.Parent = Frame

local function CreateButton(name, color, action)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Size = UDim2.new(1, -10, 0, 30)
    btn.Position = UDim2.new(0, 5, 0, 5)
    btn.BackgroundColor3 = color
    btn.BorderSizePixel = 0
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 12
    btn.Parent = ButtonFrame
    
    btn.MouseButton1Click:Connect(function()
        pcall(action)
        -- Visual feedback
        local originalColor = btn.BackgroundColor3
        btn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
        wait(0.2)
        btn.BackgroundColor3 = originalColor
    end)
    
    return btn
end

-- Create Buttons
CreateButton("Top-Up (Max)", Color3.fromRGB(0, 255, 0), function()

-- // 3. CORE FUNCTIONS //

-- Helper to find money value
local function find_money_value()
    local leaderstats = Player:FindFirstChild("leaderstats")
    if leaderstats then
        return leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Credits")
    end
    return nil
end

-- 1. Top-Up (Massive Money)
local function AddMassiveMoney()
    local money = find_money_value()
    if money then
        money.Value = money.Value + MASSIVE_MONEY_AMOUNT
        print("[Khang_Dev] Top-Up Added: " .. MASSIVE_MONEY_AMOUNT)
    end
end

-- 2. Double Money
local function DoubleMoney()
    local money = find_money_value()
    if money then
        money.Value = money.Value * DOUBLE_MULTIPLIER
        print("[Khang_Dev] Money Doubled: " .. money.Value)
    end
end

-- 3. Dupe Money (Duplicates current balance)
local function DupeMoney()
    local money = find_money_value()
    if money then
        local current_val = money.Value
        -- Dupe by adding the current value to itself
        money.Value = current_val + current_val
        print("[Khang_Dev] Money Duplicated: " .. money.Value)
    end
end

-- 4. Clone Investors
local function CloneInvestors()
    local workspace = game:GetService("Workspace")
    
    -- Define possible prefixes/names for investor/employee models
    local investor_patterns = {"Investor", "Employee", "Worker", "Hired", "Staff", "LemonSeller", "Lemon"}
    
    -- Check common folders
    local folders_to_check = {
        workspace:FindFirstChild("Shops"),
        workspace:FindFirstChild("Baseplate"),
        workspace
    }
    
    for _, folder in ipairs(folders_to_check) do
        if not folder then continue end
        
        for _, child in ipairs(folder:GetChildren()) do
            if child:IsA("Model") then
                for _, pattern in ipairs(investor_patterns) do
                    if string.find(child.Name, pattern, 1, true) then
                        pcall(function()
                            local new_investor = child:Clone()
                            new_investor.Parent = folder
                            
                            -- Offset position to prevent clipping
                            if new_investor:FindFirstChild("HumanoidRootPart") then
                                local root = new_investor.HumanoidRootPart
                                root.CFrame = root.CFrame * CFrame.new(2, 0, 0)
                            end
                        end)
                    end
                end
            end
        end
    end
    print("[Khang_Dev] Investors Cloned")
end

-- Initial execution on load
AddMassiveMoney()

DupeMoney()
