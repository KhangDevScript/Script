-- // 1. CONFIGURATION //
local CONFIG = {
    MASSIVE_MONEY = "9999999999999e+999999",
    DOUBLE_MULTIPLIER = 2,
    LOCK_NAME = "KhangDev_Lock_Perma"
}

-- // 2. STATE //
local State = {
    isLocked = false,
    guiCreated = false,
    screenGui = nil,
    label = nil,
    frame = nil
}

-- // 3. UI SYSTEM //

local function createLockUI()
    if State.guiCreated then return end
    
    local Player = game.Players.LocalPlayer
    if not Player then return end
    
    local PlayerGui = Player:FindFirstChild("PlayerGui")
    if not PlayerGui then 
        PlayerGui = Player:WaitForChild("PlayerGui", 5) 
        if not PlayerGui then return end 
    end
    
    -- Create ScreenGui
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "KhangDev_GUI"
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    ScreenGui.ResetOnSpawn = false
    ScreenGui.Parent = PlayerGui
    
    -- Create Frame
    local Frame = Instance.new("Frame")
    Frame.Name = "LockStatus"
    Frame.Size = UDim2.new(0, 240, 0, 45)
    Frame.Position = UDim2.new(0.5, -120, 0, 15)
    Frame.BackgroundTransparency = 0.15
    Frame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    Frame.BorderSizePixel = 1
    Frame.BorderColor3 = Color3.fromRGB(100, 100, 100)
    Frame.Parent = ScreenGui
    
    -- Create Label
    local Label = Instance.new("TextLabel")
    Label.Name = "StatusText"
    Label.Size = UDim2.new(1, -40, 1, -5)
    Label.Position = UDim2.new(0, 5, 0, 5)
    Label.BackgroundTransparency = 1
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.Font = Enum.Font.GothamBold
    Label.TextSize = 16
    Label.Text = "Khang_dz: Initializing..."
    Label.Parent = Frame
    
    -- Create Close Button
    local CloseBtn = Instance.new("TextButton")
    CloseBtn.Name = "CloseBtn"
    CloseBtn.Size = UDim2.new(0, 30, 0, 30)
    CloseBtn.Position = UDim2.new(1, -35, 0, 7)
    CloseBtn.Text = "X"
    CloseBtn.TextSize = 14
    CloseBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    CloseBtn.BackgroundColor3 = Color3.fromRGB(200, 0, 0)
    CloseBtn.Parent = Frame
    
    CloseBtn.MouseButton1Click:Connect(function()
        Frame:Destroy()
        ScreenGui:Destroy()
        State.guiCreated = false
        State.isLocked = false
        print("[Khang_dz] Lock released by user.")
    end)
    
    State.screenGui = ScreenGui
    State.label = Label
    State.frame = Frame
    State.guiCreated = true
    
    -- Wait for UI to render (Delta X specific)
    task.wait(0.5) 
end

local function updateStatus(text, color)
    if not State.label or not State.frame then return end
    State.label.Text = "Khang_dz: " .. text
    State.frame.BackgroundColor3 = color
end

local function acquireLock()
    if State.isLocked then
        updateStatus("Already Running", Color3.fromRGB(255, 100, 100))
        return false
    end
    
    createLockUI()
    State.isLocked = true
    updateStatus("Locked & Running", Color3.fromRGB(0, 255, 0))
    return true
end

-- // 4. CORE FUNCTIONS //

local function findMoneyValue()
    local Player = game.Players.LocalPlayer
    if not Player then task.wait(1); return findMoneyValue() end
    
    local leaderstats = Player:FindFirstChild("leaderstats")
    if leaderstats then
        local money = leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Credits")
        if money then return money end
    end
    return nil
end

local function doubleMoney()
    local moneyObj = findMoneyValue()
    if not moneyObj then return end
    
    local current = moneyObj.Value
    if typeof(current) == "number" then
        moneyObj.Value = current * CONFIG.DOUBLE_MULTIPLIER
    end
end

local function doubleInvestors()
    local Workspace = game:GetService("Workspace")
    local patterns = {"Investor", "Employee", "Worker", "Staff", "Hired", "Lemon"}
    local count = 0
    
    local foldersToSearch = {
        Workspace:FindFirstChild("Shops") or Workspace,
        Workspace:FindFirstChild("Baseplate") or Workspace
    }
    
    for _, folder in ipairs(foldersToSearch) do
        if not folder then continue end
        
        for _, child in ipairs(folder:GetChildren()) do
            if child:IsA("Model") then
                for _, pattern in ipairs(patterns) do
                    -- Fixed string.find for Delta X compatibility (Lua 5.2+)
                    if string.find(child.Name, pattern, 1, true) then
                        pcall(function()
                            local clone = child:Clone()
                            clone.Parent = folder
                            
                            if clone:FindFirstChild("HumanoidRootPart") then
                                local root = clone.HumanoidRootPart
                                local offset = CFrame.new(2, 0, 0)
                                root.CFrame = root.CFrame * offset
                            end
                            
                            count = count + 1
                        end)
                    end
                end
            end
        end
    end
end

local function injectMassiveMoney()
    local moneyObj = findMoneyValue()
    if not moneyObj then return end
    
    local massiveVal = tonumber(CONFIG.MASSIVE_MONEY)
    
    if massiveVal then
        moneyObj.Value = moneyObj.Value + massiveVal
    else
        if typeof(moneyObj.Value) == "string" then
            moneyObj.Value = CONFIG.MASSIVE_MONEY
        else
            moneyObj.Value = math.huge
        end
    end
end

-- // 5. MAIN EXECUTION //

local function main()
    if not acquireLock() then return end
    
    local success, err = pcall(function()
        updateStatus("Executing Money...", Color3.fromRGB(0, 255, 0))
        doubleMoney()
        injectMassiveMoney()
        
        updateStatus("Executing Investors...", Color3.fromRGB(0, 255, 0))
        doubleInvestors()
        
        updateStatus("Completed (Permanent Lock)", Color3.fromRGB(100, 100, 100))
    end)
    
    if not success then
        updateStatus("Error: " .. tostring(err), Color3.fromRGB(255, 100, 100))
    end
end

main()
