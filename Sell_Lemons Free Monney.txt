-- // 1. CONFIG //
local MASSIVE_AMOUNT = 9999999999999e+999999
local MULTIPLIER = 2

-- // 2. UI CREATION //
local Player = game.Players.LocalPlayer
local PlayerGui = Player:WaitForChild("PlayerGui")

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "KhangDv_UI"
ScreenGui.ResetOnSpawn = false
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
ScreenGui.Parent = PlayerGui

local Frame = Instance.new("Frame")
Frame.Name = "MainFrame"
Frame.Size = UDim2.new(0, 250, 0, 150)
Frame.Position = UDim2.new(0.5, -125, 0.1, 0)
Frame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
Frame.BorderSizePixel = 0
Frame.Parent = ScreenGui

local Header = Instance.new("TextLabel")
Header.Name = "Header"
Header.Size = UDim2.new(1, 0, 0, 30)
Header.BackgroundTransparency = 0.5
Header.Text = "Khang_Dv Control"
Header.TextColor3 = Color3.fromRGB(255, 255, 255)
Header.Font = Enum.Font.GothamBold
Header.TextSize = 14
Header.Parent = Frame

local ButtonFrame = Instance.new("Frame")
ButtonFrame.Name = "Buttons"
ButtonFrame.Size = UDim2.new(1, -20, 1, -30)
ButtonFrame.Position = UDim2.new(0, 10, 0, 30)
ButtonFrame.BackgroundTransparency = 1
ButtonFrame.Parent = Frame

local function CreateButton(name, color, action)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Size = UDim2.new(1, -10, 0, 30)
    btn.Position = UDim2.new(0, 5, 0, 5)
    btn.BackgroundColor3 = color
    btn.BorderSizePixel = 0
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 12
    btn.Parent = ButtonFrame
    
    btn.MouseButton1Click:Connect(function()
        pcall(action)
        btn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
        wait(0.2)
        btn.BackgroundColor3 = color
    end)
    
    return btn
end

CreateButton("Top-Up (Max)", Color3.fromRGB(0, 255, 0), function()
    AddMoney()
end)

CreateButton("Double Money", Color3.fromRGB(0, 150, 255), function()
    DoubleMoney()
end)

CreateButton("Clone Investors", Color3.fromRGB(255, 165, 0), function()
    CloneInvestors()
end)

-- // 3. CORE FUNCTIONS //

local function GetMoneyObj()
    local Leader = Player:FindFirstChild("leaderstats")
    if not Leader then return nil end
    return Leader:FindFirstChild("Money") or Leader:FindFirstChild("Cash") or Leader:FindFirstChild("Credits")
end

local function AddMoney()
    local Money = GetMoneyObj()
    if Money then
        Money.Value = Money.Value + MASSIVE_AMOUNT
    end
end

local function DoubleMoney()
    local Money = GetMoneyObj()
    if Money then
        Money.Value = Money.Value * MULTIPLIER
    end
end

local function CloneInvestors()
    local Workspace = game:GetService("Workspace")
    local Patterns = {"Investor", "Employee", "Worker", "Staff", "Lemon"}
    local Folders = {Workspace:FindFirstChild("Shops"), Workspace:FindFirstChild("Baseplate"), Workspace}
    
    for _, Folder in ipairs(Folders) do
        if not Folder then continue end
        for _, Child in ipairs(Folder:GetChildren()) do
            if Child:IsA("Model") then
                for _, Pat in ipairs(Patterns) do
                    if string.find(Child.Name, Pat, 1, true) then
                        pcall(function()
                            local Clone = Child:Clone()
                            Clone.Parent = Folder
                            if Clone:FindFirstChild("HumanoidRootPart") then
                                Clone.HumanoidRootPart.CFrame = Clone.HumanoidRootPart.CFrame * CFrame.new(2, 0, 0)
                            end
                        end)
                    end
                end
            end
        end
    end
end
