local CUSTOM_MONEY_VALUE = 134.803e2721 -- 134.803 x 10^2721

-- // 2. HELPER: HSL to RGB //
local function hsl_to_rgb(h, s, l)
    h = h % 1
    s = math.clamp(s, 0, 1)
    l = math.clamp(l, 0, 1)
    local q = l < 0.5 and l * (1 + s) or l + s - l * s
    local p = 2 * l - q
    local k = {p, p + 2/3, p - 2/3}
    local function color(val)
        val = val + p
        if val < 0 then val = 0 end
        if val > 1 then val = 1 end
        if val < 1/6 then return p + (q - p) * 6 * val end
        if val < 1/2 then return q end
        if val < 2/3 then return p + (q - p) * (2/3 - val) * 6 end
        return p
    end
    return Color3.fromRGB(
        math.floor(color(k[1]) * 255),
        math.floor(color(k[2]) * 255),
        math.floor(color(k[3]) * 255)
    )
end

local hue = 0

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        hue = (hue + 0.005) % 1
        local currentColor = hsl_to_rgb(hue, 1, 0.5)
        for _, btn in ipairs(ButtonContainer:GetChildren()) do
            if btn:IsA("TextButton") and btn.BackgroundColor3 ~= Color3.fromRGB(200, 0, 0) then

-- // 4. CORE FUNCTIONS //

local function find_money_value(custom_name)
    local leaderstats = Player:FindFirstChild("leaderstats")
    if leaderstats then
        -- If custom name is provided, use it
        if custom_name and custom_name ~= "" then
            local val = leaderstats:FindFirstChild(custom_name)
            if val then return val end
        end
        -- Fallback to common names
        return leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Credits")
    end
    return nil
end

local function SetCustomMoney()
    local customName = InputBox.Text
    local money = find_money_value(customName)
    if money then
        money.Value = CUSTOM_MONEY_VALUE
        print("[Khang_Dev] Money Set to: " .. CUSTOM_MONEY_VALUE)
    else
        print("[Khang_Dev] Money not found. Check the name in the text box.")
    end
end

local function DoubleMoney()
    local customName = InputBox.Text
    local money = find_money_value(customName)
    if money then
        money.Value = money.Value * DOUBLE_MULTIPLIER
        print("[Khang_Dev] Money Doubled")
    end
end

local function DupeMoney()
    local customName = InputBox.Text
    local money = find_money_value(customName)
    if money then
        money.Value = money.Value + money.Value
        print("[Khang_Dev] Money Duplicated")
    end
end

local function CloneInvestors()
    local workspace = game:GetService("Workspace")
    local investor_patterns = {"Investor", "Employee", "Worker", "Hired", "Staff", "LemonSeller", "Lemon"}
    local folders_to_check = {workspace:FindFirstChild("Shops"), workspace:FindFirstChild("Baseplate"), workspace}
    
    for _, folder in ipairs(folders_to_check) do
        if not folder then continue end
        for _, child in ipairs(folder:GetChildren()) do
            if child:IsA("Model") then
                for _, pattern in ipairs(investor_patterns) do
                    if string.find(child.Name, pattern, 1, true) then
                        pcall(function()
                            local new_investor = child:Clone()
                            new_investor.Parent = folder
                            if new_investor:FindFirstChild("HumanoidRootPart") then
                                new_investor.HumanoidRootPart.CFrame = new_investor.HumanoidRootPart.CFrame * CFrame.new(2, 0, 0)
                            end
                        end)
                    end
                end
            end
        end
    end
    print("[Khang_Dev] Investors Cloned")
end

    local btn = Instance.new("TextButton")

    btn.BorderSizePixel = 0
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 12

    btn.MouseButton1Click:Connect(function()

        btn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        btn.BackgroundColor3 = originalColor
        pcall(action)
    end)
    
    return btn
end
