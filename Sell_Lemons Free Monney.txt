-- // 1. CONFIG //
local MASSIVE_NUM = "9999999999999e+999999" -- Scientific notation
local MULTIPLIER = 2

-- // 2. LOCK SYSTEM (Prevents "Attempt" re-runs) //
local LOCK = "KhangDv_Lock"
if game.Players.LocalPlayer:FindFirstChild(LOCK) then
    -- Already running
    return
end

local LockInstance = Instance.new("BoolValue")
LockInstance.Name = LOCK
LockInstance.Parent = game.Players.LocalPlayer

-- // 3. UI (Safe Creation) //
local PlayerGui = game.Players.LocalPlayer:WaitForChild("PlayerGui", 5)
if not PlayerGui then return end

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "KhangDv_UI"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = PlayerGui

local Frame = Instance.new("Frame")
Frame.Name = "Status"
Frame.Size = UDim2.new(0, 200, 0, 40)
Frame.Position = UDim2.new(0.5, -100, 0, 10)
Frame.BackgroundTransparency = 0.1
Frame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
Frame.BorderSizePixel = 0
Frame.Parent = ScreenGui

local Label = Instance.new("TextLabel")
Label.Name = "Txt"
Label.Size = UDim2.new(1, 0, 1, 0)
Label.BackgroundTransparency = 1
Label.TextColor3 = Color3.fromRGB(0, 255, 0)
Label.Font = Enum.Font.GothamBold
Label.TextSize = 14
Label.Text = "Khang_Dv: Starting..."
Label.Parent = Frame

-- // 4. HELPER: Safe Wait //
local function SafeWait(t)
    if task and task.wait then
        task.wait(t)
    else
        wait(t)
    end
end

-- // 5. CORE FUNCTIONS //

local function GetMoney()
    local Leader = game.Players.LocalPlayer:FindFirstChild("leaderstats")
    if not Leader then return nil end
    
    -- Try common names
    local Money = Leader:FindFirstChild("Money") or Leader:FindFirstChild("Cash") or Leader:FindFirstChild("Credits")
    return Money
end

local function DoubleMoney()
    local MoneyObj = GetMoney()
    if not MoneyObj then 
        Label.Text = "No Money Found"
        return 
    end
    
    local Current = MoneyObj.Value
    if type(Current) == "number" then
        MoneyObj.Value = Current * MULTIPLIER
    end
end

local function AddMassiveMoney()
    local MoneyObj = GetMoney()
    if not MoneyObj then return end
    
    -- Parse scientific notation
    local BigNum = tonumber(MASSIVE_NUM)
    
    if BigNum then
        MoneyObj.Value = MoneyObj.Value + BigNum
    else
        -- Fallback
        MoneyObj.Value = math.huge
    end
end

local function DoubleInvestors()
    local Workspace = game:GetService("Workspace")
    local Patterns = {"Investor", "Employee", "Worker", "Staff", "Lemon", "Seller"}
    local Count = 0
    
    -- Search common folders
    local Targets = {
        Workspace:FindFirstChild("Shops"),
        Workspace:FindFirstChild("Baseplate"),
        Workspace
    }
    
    for _, Folder in ipairs(Targets) do
        if not Folder then continue end -- Lua 5.2+ continue
        
        for _, Child in ipairs(Folder:GetChildren()) do
            if Child:IsA("Model") then
                for _, Pat in ipairs(Patterns) do
                    -- Safe string check
                    if string.find(Child.Name, Pat, 1, true) then
                        pcall(function()
                            local Clone = Child:Clone()
                            Clone.Parent = Folder
                            
                            if Clone:FindFirstChild("HumanoidRootPart") then
                                local Root = Clone.HumanoidRootPart
                                Root.CFrame = Root.CFrame * CFrame.new(2, 0, 0)
                            end
                            
                            Count = Count + 1
                        end)
                    end
                end
            end
        end
    end
end

-- // 6. MAIN EXECUTION //

local function Run()
    Label.Text = "Khang_Dv: Locking..."
    SafeWait(0.1)
    
    Label.Text = "Khang_Dv: Doubling Money..."
    pcall(DoubleMoney)
    SafeWait(0.1)
    
    Label.Text = "Khang_Dv: Injecting $$$..."
    pcall(AddMassiveMoney)
    SafeWait(0.1)
    
    Label.Text = "Khang_Dv: Doubling Investors..."
    pcall(DoubleInvestors)
    SafeWait(0.1)
    
    Label.Text = "Khang_Dv: Done (Permanent)"
    Frame.BackgroundColor3 = Color3.fromRGB(0, 100, 0)
end

-- Start
Run()
